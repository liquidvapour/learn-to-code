; SOS beacon with output: copy this program into the editor and run it.
; Memory 0-2 holds SOS; each of three flashes writes SOS and ... on separate lines.
; Jump targets are zero-based instruction addresses, not source lines.
MOV R0, 83     ; S
STORE R0, 0
PUSH R0        ; Save S while preparing the middle letter
MOV R0, 79     ; O
STORE R0, 1
POP R0         ; Restore S from the stack
STORE R0, 2
MOV R0, 46     ; '.'
STORE R0, 4
MOV R1, 1      ; Decrement amount
MOV R2, 3      ; Flashes remaining
; PC 11: output one flash
LOAD R3, 0
OUT R3         ; S
LOAD R3, 1
OUT R3         ; O
LOAD R3, 2
OUT R3         ; S
MOV R3, 10
OUT R3         ; Newline
OUT R0         ; .
OUT R0         ; .
OUT R0         ; .
OUT R3         ; Newline
SUB R2, R1
JZ 26          ; Zero remaining: stop
JMP 11         ; Flashes remain: repeat
; PC 26: stop
HALT
